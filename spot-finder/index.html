🔘 Cara Menggunakan Alat Counter Wireless:

  • Tanpa Kabel Layar Tambahan: Cukup board ESP32-C3 SuperMini dicolok daya (USB / Baterai / Powerbank).
  • Tekan Tombol Fisik BOOT (GPIO 9): Setiap kali tombol ditekan, LED onboard berkedip & jumlah orang di Gazebo bertambah +1.
  • Sinkron Otomatis: Website SpotFinder IT Del langsung memperbarui sisa kursi secara real-time.
#include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <ArduinoJson.h>

// --- DEFINISI LAYAR OLED SSD1306 128x64 ---
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1
#define SCREEN_ADDRESS 0x3C // Alamat I2C OLED (0x3C atau 0x3D)

// PIN I2C DEFAULT ESP32-C3 SUPERMINI
#define I2C_SDA 8
#define I2C_SCL 9

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// --- KONFIGURASI WIFI & MQTT ---
const char* ssid        = "NAMA_WIFI_ANDA";
const char* password    = "PASSWORD_WIFI";
const char* mqtt_server = "76.13.19.250"; // IP Broker MQTT
const int   mqtt_port   = 1883;
const char* mqtt_topic  = "itdel/gazebo/status";

WiFiClient espClient;
PubSubClient client(espClient);

void renderOled(String title, int kosong, int total, int persen) {
  display.clearDisplay();
  
  // Header bar
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.print(F("SPOTFINDER IT DEL"));
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  // Nama Spot
  display.setCursor(0, 15);
  display.setTextSize(1);
  display.print(title.substring(0, 20));

  // Angka Ketersediaan Besar
  display.setCursor(0, 28);
  display.setTextSize(2);
  display.print(kosong);
  display.setTextSize(1);
  display.print("/");
  display.print(total);
  display.print(" KOSONG");

  // Status Bar Bawah
  display.drawLine(0, 50, 128, 50, SSD1306_WHITE);
  display.setCursor(0, 54);
  display.setTextSize(1);
  display.print("Isi: ");
  display.print(persen);
  display.print("% [MQTT:OK]");

  display.display();
}

void setup_wifi() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 10);
  display.println("Connecting WiFi...");
  display.display();

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    display.print(".");
    display.display();
  }
  
  display.clearDisplay();
  display.setCursor(0, 10);
  display.println("WiFi Terhubung!");
  display.println(WiFi.localIP());
  display.display();
  delay(1000);
}

void callback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (int i = 0; i < length; i++) message += (char)payload[i];
  
  StaticJsonDocument<256> doc;
  if (!deserializeJson(doc, message)) {
    const char* ruangan = doc["ruangan"];
    int kosong = doc["kosong"];
    int total = doc["total"];
    int persen = doc["persen"];

    renderOled(String(ruangan), kosong, total, persen);
  }
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32C3_Gazebo-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      client.subscribe(mqtt_topic);
      display.setCursor(0, 54);
      display.print("MQTT Connected!");
      display.display();
    } else {
      delay(2000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  
  // Inisialisasi I2C ESP32-C3 SuperMini (SDA=GPIO 8, SCL=GPIO 9)
  Wire.begin(I2C_SDA, I2C_SCL);
  
  if (!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for (;;);
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 10);
  display.println("ESP32-C3 SUPERMINI");
  display.println("Gazebo OLED Display");
  display.println("MQTT: 76.13.19.250");
  display.display();
  delay(1500);

  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();
}
